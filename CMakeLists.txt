cmake_minimum_required(VERSION 3.16)
set(NABTOOLS_SIG_DATA_DIR "${CMAKE_CURRENT_LIST_DIR}/src/catalogs/sig" CACHE PATH
    "Directory containing the locally generated sig.generated.inc")

function(nabtools_use_sig_data target)
  if(NOT EXISTS "${NABTOOLS_SIG_DATA_DIR}/sig.generated.inc")
    message(FATAL_ERROR "NaBTools SIG tables are missing in ${NABTOOLS_SIG_DATA_DIR}. "
        "Initialize the bluetooth-SIG submodule and run python tools/generate_sig.py, "
        "or choose the OFF profile.")
  endif()
  target_include_directories(${target} PRIVATE "${NABTOOLS_SIG_DATA_DIR}")
endfunction()
